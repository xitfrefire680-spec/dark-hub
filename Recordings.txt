tbl2.isPlayerInGroundArea = function(pos)
    local g = workspace:FindFirstChild("World") and workspace.World:FindFirstChild("Areas") and workspace.World.Areas:FindFirstChild("Ground")
    if not g then return false end
    if g:IsA("Model") then
        local bb, sz = g:GetBoundingBox()
        local lp = bb:PointToObjectSpace(pos.Position)
        if math.abs(lp.X) > sz.X/2 then return false end
        if math.abs(lp.Y) > sz.Y/2 + 20 then return false end
        if math.abs(lp.Z) > sz.Z/2 then return false end

    if g:IsA("BasePart") then
        local lp = g.CFrame:PointToObjectSpace(pos.Position)
        if math.abs(lp.X) > g.Size.X/2 then return false end
        if math.abs(lp.Y) > g.Size.Y/2 + 20 then return false end
        if math.abs(lp.Z) > g.Size.Z/2 then return false end
        return true
    end
    return false
end

tbl2.getGroundAdjustedPosition = function(pos)
    if typeof(pos) ~= "Vector3" then return pos end
    local c = localPlayer.Character or v
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    local h = c and c:FindFirstChildOfClass("Humanoid")
    local off = tonumber(tbl3.GROUND_CLEARANCE) or 4
    if hrp and h then off = math.max(off, hrp.Size.Y * 0.5 + math.max(tonumber(h.HipHeight) or 0, 0) + 0.2) end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.IgnoreWater = true
    local filt = {}
    if v and v.Parent then table.insert(filt, v) end
    local origin = pos + Vector3.new(0, tbl3.RAYCAST_UP, 0)
    local dir = Vector3.new(0, -(tbl3.RAYCAST_UP + tbl3.RAYCAST_DOWN), 0)
    for _ = 1, 8 do
        rp.FilterDescendantsInstances = filt
        local hit = workspace:Raycast(origin, dir, rp)
        if not hit then return pos end
        local inst = hit.Instance
        if inst:IsA("Terrain") or (inst:IsA("BasePart") and inst.CanCollide) then
            return Vector3.new(pos.X, hit.Position.Y + off, pos.Z)
        end
        table.insert(filt, inst)
    end
    return pos
end

tbl2.tweenToTarget = function(target, faceDir)
    local c = localPlayer.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    local h = c and c:FindFirstChildOfClass("Humanoid")
    if not hrp then return end
    if tbl2.isPlayerInGroundArea(hrp) then tbl2.unequipTreadmill() end
    local spd = tonumber(tbl3.TWEEN_SPEED) or 500
    tbl2.stopCurrentTween(false)
    if h then
        if h.WalkSpeed > 0 then walkSpeed = h.WalkSpeed end
        h.WalkSpeed = 0
    end
    local function tween1(p, cb)
        local mag = (p - hrp.Position).Magnitude
        if mag < 2 then if cb then cb() end return end
        local ti = TweenInfo.new(mag / spd, Enum.EasingStyle.Linear)
        local cf = CFrame.new(p)
        if faceDir then
            local d = Vector3.new(p.X - hrp.Position.X, 0, p.Z - hrp.Position.Z)
            if d.Magnitude > 0.01 then cf = CFrame.lookAt(p, p + d.Unit) end
        end
        local tw = TweenService:Create(hrp, ti, { CFrame = cf })
        v6 = tw
        connection = tw.Completed:Connect(function(ps)
            if v6 == tw then
                v6 = nil
                if connection then connection:Disconnect() connection = nil end
                if ps == Enum.PlaybackState.Completed then
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    hrp.CFrame = cf
                    if cb then cb() end
                end
            end
        end)
        tw:Play()
    end
    local wp = tbl3.WAYPOINT
    local atWp = (target - wp).Magnitude < 2
    local wpPos = tbl2.getGroundAdjustedPosition(wp)
    local dest = faceDir and target or tbl2.getGroundAdjustedPosition(target)
    if tbl3.USE_GROUND_DETECTION and not faceDir and tbl2.isPlayerInGroundArea(hrp) and not atWp then
        tween1(wpPos, function() tween1(dest, tbl2.completeMove) end)
    else
        tween1(dest, tbl2.completeMove)
    end
end

tbl2.getScaleMultiplier = function(arg)
    local n = tonumber(arg) or 1
    if n <= 5 then return n ^ 1.85 end
    return (n / 5) ^ 1.2 * 19.637875755794113
end

tbl2.getEggData = function(rec)
    if not Assets or not Assets.Directory then return nil end
    local info = Assets.Directory[rec.AssetCategory]
    if not info then return nil end
    local rate = tonumber(info.EarningRate) or 0
    local scale = tonumber(rec.AssetScale) or 1
    local mult = 1
    if rec.Mutations and #rec.Mutations > 0 and Mutations and Mutations.EarningsFor then
        local ok, r = pcall(Mutations.EarningsFor, rec.Mutations)
        if ok then mult = tonumber(r) or 1 end
    end
    local gen = rate * tbl2.getScaleMultiplier(scale) * mult
    local color = Color3.fromRGB(170, 190, 178)
    local rnum, rname = 0, ""
    if info.Rarity then
        rnum = info.Rarity.RarityNumber or 0
        rname = info.Rarity.DisplayName or info.Rarity.Name or ""
        color = tbl2.getOfficialRarityColor(rname)
    end
    return {
        Uid = rec.Uid,
        Category = rec.AssetCategory,
        Name = info.DisplayName or rec.AssetCategory,
        Generation = math.round(gen),
        Position = rec.BottomCFrame and rec.BottomCFrame.Position or nil,
        BottomCFrame = rec.BottomCFrame,
        Rarity = rnum, RarityName = rname, RarityColor = color,
        BaseRate = rate, Scale = scale,
    }
end

tbl2.getGroundPosition = function()
    local objs = workspace:FindFirstChild("World")
    if not objs then return nil end
    local areas = objs:FindFirstChild("Areas")
    if not areas then return nil end
    local g = areas:FindFirstChild("Ground")
    if not g then return nil end
    if g:IsA("BasePart") then
        return g.Position + Vector3.new(0, g.Size.Y * 0.5 + tbl3.GROUND_OFFSET, 0)
    end
    if g:IsA("Model") then
        local pv = g:GetPivot()
        local _, sz = g:GetBoundingBox()
        return pv.Position + Vector3.new(0, sz.Y * 0.5 + tbl3.GROUND_OFFSET, 0)
    end
    local bp = g:FindFirstChildWhichIsA("BasePart", true)
    if bp then return bp.Position + Vector3.new(0, bp.Size.Y * 0.5 + tbl3.GROUND_OFFSET, 0) end
    return nil
end

tbl2.refreshEggs = function(force)
    local now = tick()
    if not force and now - tbl4.lastRefresh < tbl3.REFRESH_INTERVAL then return tbl4.eggs end
    local ok, result = pcall(function() return tbl.AskFieldEggSnapshot:InvokeServer() end)
    if not ok then return tbl4.eggs end
    local eggs = tbl4.eggs
    table.clear(eggs)
    if result and result.Records then
        for _, rec in ipairs(result.Records) do
            if rec.State == "Slot" or rec.State == "Dropped" then
                local d = tbl2.getEggData(rec)
                if d then eggs[#eggs + 1] = d end
            end
        end
    end
    table.sort(eggs, function(a, b) return a.Generation > b.Generation end)
    local filt = tbl4.filteredEggs
    table.clear(filt)
    for _, e in ipairs(eggs) do
        if tbl2.eggMatchesRarityFilter(e) then filt[#filt + 1] = e end
    end
    tbl4.bestEgg = filt[1]
    tbl4.lastRefresh = now
    return tbl4.eggs
end

tbl2.getActiveTargetEgg = function()
    if tbl4.selectedEgg then
        for _, e in ipairs(tbl4.filteredEggs) do
            if e.Uid == tbl4.selectedEgg.Uid then
                tbl4.selectedEgg = e
                tbl4.selectedUid = e.Uid
                return e
            end
        end
        tbl4.selectedEgg = nil
        tbl4.selectedUid = nil
    end
    return tbl4.bestEgg
end

tbl2.startEggProtection = function(uid)
    if not uid then return end
    if tbl4.isProtecting and tbl4.heldEggUid == uid then return end
    tbl4.isProtecting = true
    tbl4.heldEggUid = uid
    tbl4.lastProtectTime = 0
    task.spawn(function()
        while tbl4.isProtecting and tbl4.heldEggUid == uid do
            tbl4.lastProtectTime = tick()
            local ok, r = pcall(function()
                return tbl.AskFieldEggCarry:InvokeServer({ Uid = uid })
            end)
            if ok and r == true then tbl4.carrying = true end
            task.wait(tbl3.PROTECT_INTERVAL)
        end
    end)
end

tbl2.startSpammingCarry = function(uid)
    if not uid or tbl4.isStealing then return end
    if tbl4.carrying and tbl4.heldEggUid == uid then return end
    tbl4.isStealing = true
    tbl4.lastSpamTime = 0
    task.spawn(function()
        while true do
            if tbl4.isStealing and not tbl4.carrying then
                local now = tick()
                if now - tbl4.lastSpamTime >= tbl3.SPAM_INTERVAL then
                    tbl4.lastSpamTime = now
                    local ok, r = pcall(function()
                        return tbl.AskFieldEggCarry:InvokeServer({ Uid = uid })
                    end)
                    if ok and r == true then
                        tbl4.isStealing = false
                        tbl4.carrying = true
                        tbl4.heldEggUid = uid
                        tbl4.carryLocked = true
                        pcall(tbl2.spawnBossForEgg, tbl4.targetEgg or tbl2.getActiveTargetEgg())
                        task.spawn(function() tbl2.startEggProtection(uid) end)
                        task.spawn(function()
                            local gp = tbl2.getGroundPosition()
                            local dp = tbl2.getBossDeliveryPosition(gp)
                            if gp and dp then
                                tbl4.isDelivering = true
                                tbl2.clearBeam()
                                tbl2.createBeam(gp)
                                tbl2.startMove(dp, function()
                                    tbl4.isDelivering = false
                                    tbl2.cleanupBossMount()
                                    tbl2.resetCarryState()
                                    tbl2.completeGuidedCycle()
                                end, true)
                            end
                        end)
                        break
                    end
                end
            end
            RunService.Heartbeat:Wait()
        end
        tbl4.isStealing = false
    end)
end

tbl2.startDelivery = function()
    if tbl4.isDelivering then return end
    local gp = tbl2.getGroundPosition()

    if not gp or not dp then return end
    tbl4.isDelivering = true

        local d = Vector3.new(dp.X - humanoidRootPart.Position.X, 0, dp.Z - humanoidRootPart.Position.Z)
        if d.Magnitude > 0.01 then
            humanoidRootPart.CFrame = CFrame.lookAt(humanoidRootPart.Position, humanoidRootPart.Position + d.Unit)
        end
    end
    tbl2.startMove(dp, function()
        tbl4.isDelivering = false
        tbl2.cleanupBossMount()
        tbl2.resetCarryState()
        tbl2.completeGuidedCycle()
    end, true)
end

    end

    if tbl4.mode == "TELEGUIADO" then
        tbl4.guidedCycleActive = true
        if tbl4.enabled and tbl2.updateToggle then tbl2.updateToggle() end
    end
    tbl4.targetEgg = target
    local pos = target.Position
    tbl2.createBeam(pos)
    if (pos - humanoidRootPart.Position).Magnitude <= tbl3.SPAWN_OFFSET then
        tbl2.startSpammingCarry(target.Uid)
        return true

    end)
    return true
end

        tbl4.carrying = false

                if tbl4.targetEgg and tbl4.targetEgg.Uid == uid and not tbl4.carrying then
                    tbl2.startSpammingCarry(uid)

tbl2.formatNumber = function(arg)
    local n = tonumber(arg) or 0
    local function fmt(v)
        return (string.format("%.2f", v):gsub("(%..-)0+$", "%1"):gsub("%.$", ""))
    end
    local abs = math.abs(n)
    for _, pair in ipairs({ { 1e12, "T" }, { 1e9, "B" }, { 1000000, "M" }, { 1000, "K" } }) do
        if abs >= pair[1] then return fmt(n / pair[1]) .. pair[2] end
    end
    return tostring(math.floor(n + 0.5))
end

        if vp then vp:Destroy() end
    end
end

tbl2.refreshDropdownRows = function()
    for i = 1, tbl3.DROPDOWN_ROWS do
        local ui = tbl12[i]
        local egg = tbl4.filteredEggs[i]
        ui.Egg = egg
        local lastUid = egg and tostring(egg.Uid) or nil
        local category = egg and egg.Category or nil
        if ui.LastUid ~= lastUid or ui.LastCategory ~= category then
            for _, child in ipairs(ui.Preview:GetChildren()) do child:Destroy() end
            ui.LastUid = lastUid
            ui.LastCategory = category
            if egg then tbl2.attachViewport(ui.Preview, category) end
        end
        if egg then
            ui.Row.Visible = true
            ui.NameLabel.Text = tostring(egg.Name or "UNKNOWN")
            ui.RarityLabel.Text = tostring(egg.RarityName or "")
            ui.RarityLabel.TextColor3 = egg.RarityColor or C.textMuted
            ui.ValueLabel.Text = tbl2.formatNumber(egg.Generation or 0)
        else
            ui.Row.Visible = false
            ui.NameLabel.Text = ""
            ui.RarityLabel.Text = ""
            ui.ValueLabel.Text = ""
        end
    end
end
